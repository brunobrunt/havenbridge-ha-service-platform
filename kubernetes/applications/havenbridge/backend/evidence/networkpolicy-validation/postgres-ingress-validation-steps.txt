2. Describe the policy
----------------------
ssh -i /home/alabi/.ssh/eph_k8s \
  mino@172.16.10.31 \
  'kubectl describe networkpolicy \
    allow-havenbridge-api-to-postgres \
    --namespace havenbridge'

Confirm:
- destination Pods are PostgreSQL;
- policy type is Ingress;
- source namespace is havenbridge;
- source Pod label is app.kubernetes.io/name=havenbridge-api;
- allowed port is TCP/5432.

4. Negative-test manifest
-------------------------
Create:
/home/alabi/projects/havenbridge-ha-service-platform/kubernetes/applications/havenbridge/backend/evidence/networkpolicy-validation/postgres-unauthorized-client.yaml

Use:

apiVersion: v1
kind: Pod
metadata:
  name: postgres-unauthorized-client
  namespace: default
  labels:
    app.kubernetes.io/name: postgres-unauthorized-client
spec:
  restartPolicy: Never
  automountServiceAccountToken: false
  securityContext:
    runAsNonRoot: true
    runAsUser: 1000
    runAsGroup: 1000
    seccompProfile:
      type: RuntimeDefault
  containers:
    - name: client
      image: busybox:1.36.1
      imagePullPolicy: IfNotPresent
      command:
        - sh
        - -c
      args:
        - |
          echo "Testing unauthorized PostgreSQL access"
          if nc -z -w 5 havenbridge-postgres.havenbridge.svc.cluster.local 5432
          then
            echo "UNEXPECTED: PostgreSQL connection was allowed."
            exit 1
          else
            echo "EXPECTED: PostgreSQL ingress NetworkPolicy blocked the connection."
            exit 0
          fi
      securityContext:
        allowPrivilegeEscalation: false
        readOnlyRootFilesystem: true
        capabilities:
          drop:
            - ALL
